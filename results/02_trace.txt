   3: import numpy as np
   4: from psi_continuum_v2.cosmology.models.psicdm_params import PsiCDMParams
   5: from psi_continuum_v2.cosmology.constants import C_LIGHT
   6: 
   7: 
   8: def E_psicdm(z: np.ndarray, params: PsiCDMParams) -> np.ndarray:
   9:     """
  10:     The simplest phenomenological Ψ-deformation of ΛCDM:
  11: 
  12:     E^2(z) = [ Ω_m (1+z)^3 + (1-Ω_m) + (1-Ω_m)*eps0*(1+z)^n ] /
  13:              [ 1 + (1-Ω_m)*eps0 ]
  14: 
  15:     Normalized so that E(0) = 1, and reduces to ΛCDM when eps0 = 0.
  16:     """
  17:     z = np.asarray(z, dtype=float)
  18:     Om0 = params.Om0
  19:     eps0 = params.eps0
  20:     n = params.n
  21: 
  22:     base = Om0 * (1.0 + z) ** 3 + (1.0 - Om0)
  23:     psi_term = (1.0 - Om0) * eps0 * (1.0 + z) ** n
  24: 
  25:     norm0 = 1.0 + (1.0 - Om0) * eps0  # ensures E(0)=1
  26: 
  27:     E2 = (base + psi_term) / norm0
  28:     return np.sqrt(E2)
  29: 
  30: 
  31: def H_psicdm(z: np.ndarray, params: PsiCDMParams) -> np.ndarray:
  32:     """
  33:     H(z) in km/s/Mpc for ΨCDM.
  34:     """
  35:     return params.H0 * E_psicdm(z, params)
  36: 
  37: 
  38: def dL_psicdm(z: np.ndarray, params: PsiCDMParams, nz: int = 4000) -> np.ndarray:
  39:     """
  40:     Luminosity distance d_L(z) in Mpc for ΨCDM via numerical integral:
  41: 
  42:         d_c(z) = c/H0 ∫ dz'/E(z')
  43:         d_L = (1 + z) * d_c
  44:     """
  45:     z = np.asarray(z, dtype=float)
  46:     z_max = float(np.max(z))
  47:     if z_max <= 0.0:
  48:         return np.zeros_like(z)
  49: 
  50:     z_grid = np.linspace(0.0, z_max, nz)
  51:     dz = z_grid[1] - z_grid[0]
  52: 
  53:     inv_E = 1.0 / E_psicdm(z_grid, params)
  54: 
  55:     # Cumulative trapezoidal integration
  56:     I = np.zeros_like(z_grid)
  57:     for i in range(1, len(z_grid)):
  58:         I[i] = I[i - 1] + 0.5 * (inv_E[i - 1] + inv_E[i]) * dz
  59: 
  60:     # d_c(z)
  61:     d_c_grid = (C_LIGHT / params.H0) * I
  62:     d_c = np.interp(z, z_grid, d_c_grid)
  63: 
  64:     return (1.0 + z) * d_c
  65: 
  66: 
  67: # --- BAO distances for ΨCDM ---
  68:
